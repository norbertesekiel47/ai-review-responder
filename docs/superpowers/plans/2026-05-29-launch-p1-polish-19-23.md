- [ ] **Step 3: Add the cleanup to the useEffect**

Replace the `useEffect` body in `src/app/admin/payment-success/page.tsx` with the cancellation-guarded version (mirrors the onboarding `PaymentSuccessClient` pattern):

```tsx
  useEffect(() => {
    if (!sessionId) {
      setStatus('error');
      setError('No session ID found in URL');
      return;
    }

    let cancelled = false;
    let pollTimer: ReturnType<typeof setTimeout> | null = null;
    let redirectTimer: ReturnType<typeof setTimeout> | null = null;
    let attempts = 0;
    const maxAttempts = 15;

    async function checkBusiness() {
      if (cancelled) return;
      try {
        const res = await fetch(`/api/admin/payment-success?session_id=${sessionId}`);
        const json = await res.json();
        if (cancelled) return;

        if (res.ok && json.businessId) {
          setBusinessId(json.businessId);
          setStatus('success');
          // Auto-redirect after 3 seconds
          redirectTimer = setTimeout(() => {
            if (cancelled) return;
            router.push(`/admin/businesses/${json.businessId}`);
          }, 3000);
          return;
        }

        attempts++;
        if (attempts < maxAttempts) {
          pollTimer = setTimeout(checkBusiness, 2000);
        } else {
          setStatus('error');
          setError('Could not find the business. The payment may still be processing — check the admin overview.');
        }
      } catch {
        if (cancelled) return;
        setStatus('error');
        setError('Network error — check the admin overview for your business.');
      }
    }

    void checkBusiness();

    return () => {
      cancelled = true;
```
