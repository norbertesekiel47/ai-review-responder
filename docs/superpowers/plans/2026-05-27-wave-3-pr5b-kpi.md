## KPI definitions (single source of truth)

All windowed metrics use a 30-day window unless noted. "Received in window" = `reviews.detected_at >= NOW() - INTERVAL '30 days'` (detection time is always set and monotonic; `review_date` can be NULL or far in the past for backfilled rows). Always exclude `status = 'deleted'`.

| KPI | Definition |
|---|---|
| `responseRate` | `{ handled, received, ratePct }`. `received` = reviews detected in last 30d (not deleted). `handled` = those received reviews that have at least one response with status ∈ `HANDLED_STATUSES`. `ratePct` = `round(handled/received*100)`, or `null` when `received = 0`. |
| `averageStarRating` | `AVG(star_rating)` over reviews detected in last 30d (not deleted, `star_rating IS NOT NULL`); `null` when none. Rounded to 2 decimals. |
| `reviewVolume` | `{ currentMonth, priorMonth, deltaPct }`. Counts of non-deleted reviews by `detected_at` in the current vs prior calendar month (`DATE_TRUNC('month', NOW())`). `deltaPct` = `round((currentMonth - priorMonth)/priorMonth*100)`, or `null` when `priorMonth = 0`. |
| `averageTimeToRespondHours` | Over responses with status ∈ `HANDLED_STATUSES`, `approved_at IS NOT NULL`, `approved_at >= NOW()-30d`, joined to their review: `AVG(EXTRACT(EPOCH FROM (approved_at - reviews.detected_at)))/3600`, rounded to 1 decimal; `null` when none. (Measures owner-action latency from detection; `detected_at <= approved_at` so no negatives. This intentionally uses `detected_at` rather than the spec's looser "review date" for robustness — backfilled `review_date` would skew it.) |
| `sentimentTrend` | 8-week ascending series, week-bucketed by `DATE_TRUNC('week', detected_at)`, zero/null-filled via `generate_series`. Each point: `{ weekStart (ISO date), averageScore (AVG(sentiment_score), null when no reviews that week), reviewCount }`. |

`HANDLED_STATUSES` (from `src/lib/shared/response-status.ts`) = `['approved','edited','awaiting_owner_connect','posted']`.

```ts
// Launch KPI summary (Spec 11). Distinct from dashboard valueSummary (lifetime
// metrics) — these are windowed (last 30 days / calendar-month / 8-week trend),
// scoped per-location (business_id) or per-account (all active locations).

export interface KpiResponseRate {
  /** Reviews received in the window that have a handled response. */
  handled: number;
  /** Reviews received in the window (not deleted). */
  received: number;
  /** round(handled/received*100); null when received = 0. */
  ratePct: number | null;
}

export interface KpiReviewVolume {
  currentMonth: number;
  priorMonth: number;
  /** round((current-prior)/prior*100); null when prior = 0. */
  deltaPct: number | null;
}

export interface SentimentTrendPoint {
  /** ISO date (week start, Monday per DATE_TRUNC('week', ...)). */
  weekStart: string;
  /** AVG(sentiment_score) that week; null when no reviews. */
  averageScore: number | null;
  reviewCount: number;
}

export type KpiScope =
  | { kind: 'location'; businessId: string }
  | { kind: 'account'; accountId: string };

export interface DashboardKpiSummary {
  scope: KpiScope;
  /** Rolling window for windowed metrics. */
  windowDays: number;
  responseRate: KpiResponseRate;
  /** AVG(star_rating) over the window, 2 decimals; null when none. */
  averageStarRating: number | null;
  reviewVolume: KpiReviewVolume;
  /** Owner-action latency (approved_at - detected_at) in hours, 1 decimal; null when none. */
  averageTimeToRespondHours: number | null;
  /** 8-week ascending sentiment series. */
  sentimentTrend: SentimentTrendPoint[];
}

export const KPI_WINDOW_DAYS = 30;
export const SENTIMENT_TREND_WEEKS = 8;
```

```ts
// src/lib/kpi/service.ts
//
// Launch KPI layer (Spec 11). Windowed metrics scoped per-location (business_id)
// or per-account (active locations). Mirrors dashboard/service.ts query style.
// Does NOT overload valueSummary (lifetime). No real-DB test — keep SQL standard.

import { sql } from '@/lib/db/server';
import { HANDLED_STATUSES } from '@/lib/shared/response-status';
import {
  DashboardKpiSummary,
  KPI_WINDOW_DAYS,
  KpiScope,
} from './types';

const HANDLED = [...HANDLED_STATUSES] as string[];

function round(value: number | null, decimals: number): number | null {
  if (value == null || Number.isNaN(value)) return null;
  const f = 10 ** decimals;
  return Math.round(value * f) / f;
}

function ratePct(handled: number, received: number): number | null {
  return received > 0 ? Math.round((handled / received) * 100) : null;
}

function deltaPct(current: number, prior: number): number | null {
  return prior > 0 ? Math.round(((current - prior) / prior) * 100) : null;
}

function mapTrend(
  rows: Array<{ week_start: unknown; avg_score: unknown; cnt: unknown }>,
): DashboardKpiSummary['sentimentTrend'] {
  return rows.map((r) => ({
    weekStart: String(r.week_start),
    averageScore: r.avg_score == null ? null : round(Number(r.avg_score), 2),
    reviewCount: Number(r.cnt ?? 0),
  }));
}

export async function getLocationKpiSummary(
  businessId: string,
): Promise<DashboardKpiSummary> {
  const [rateRows, starRows, volumeRows, ttrRows, trendRows] = await Promise.all([
    // 0: response rate (received in window + handled subset)
    sql<{ received: number; handled: number }[]>`
      SELECT
        COUNT(*)::int AS received,
        COUNT(*) FILTER (
          WHERE EXISTS (
            SELECT 1 FROM responses resp
            WHERE resp.review_id = r.id
              AND resp.status = ANY(${HANDLED}::text[])
          )
        )::int AS handled
      FROM reviews r
      WHERE r.business_id = ${businessId}
        AND r.status <> 'deleted'
        AND r.detected_at >= NOW() - INTERVAL '30 days'
    `,
    // 1: average star rating (window)
    sql<{ avg: number | null }[]>`
      SELECT AVG(star_rating)::float AS avg
      FROM reviews
      WHERE business_id = ${businessId}
        AND status <> 'deleted'
        AND star_rating IS NOT NULL
        AND detected_at >= NOW() - INTERVAL '30 days'
    `,
    // 2: review volume MoM (calendar months by detected_at)
    sql<{ current_month: number; prior_month: number }[]>`
      SELECT
        COUNT(*) FILTER (
          WHERE detected_at >= DATE_TRUNC('month', NOW())
        )::int AS current_month,
        COUNT(*) FILTER (
          WHERE detected_at >= DATE_TRUNC('month', NOW()) - INTERVAL '1 month'
            AND detected_at < DATE_TRUNC('month', NOW())
        )::int AS prior_month
      FROM reviews
      WHERE business_id = ${businessId}
        AND status <> 'deleted'
    `,
    // 3: average time-to-respond hours (approved_at - detected_at)
    sql<{ avg_hours: number | null }[]>`
      SELECT AVG(EXTRACT(EPOCH FROM (resp.approved_at - r.detected_at)) / 3600.0)::float AS avg_hours
      FROM responses resp
      JOIN reviews r ON r.id = resp.review_id
      WHERE r.business_id = ${businessId}
        AND resp.status = ANY(${HANDLED}::text[])
        AND resp.approved_at IS NOT NULL
        AND resp.approved_at >= NOW() - INTERVAL '30 days'
    `,
    // 4: weekly sentiment trend, last 8 weeks, zero/null-filled
    sql<{ week_start: string; avg_score: number | null; cnt: number }[]>`
      SELECT gs::date AS week_start,
             agg.avg_score::float AS avg_score,
             COALESCE(agg.cnt, 0)::int AS cnt
      FROM generate_series(
        DATE_TRUNC('week', NOW() - INTERVAL '7 weeks'),
        DATE_TRUNC('week', NOW()),
        INTERVAL '1 week'
      ) gs
      LEFT JOIN (
        SELECT DATE_TRUNC('week', detected_at)::date AS wk,
               AVG(sentiment_score) AS avg_score,
               COUNT(*) AS cnt
        FROM reviews
        WHERE business_id = ${businessId}
          AND status <> 'deleted'
          AND detected_at >= NOW() - INTERVAL '8 weeks'
        GROUP BY DATE_TRUNC('week', detected_at)
      ) agg ON agg.wk = gs::date
      ORDER BY week_start ASC
    `,
  ]);

  const received = rateRows[0]?.received ?? 0;
  const handled = rateRows[0]?.handled ?? 0;
  const current = volumeRows[0]?.current_month ?? 0;
  const prior = volumeRows[0]?.prior_month ?? 0;
  const scope: KpiScope = { kind: 'location', businessId };

  return {
    scope,
    windowDays: KPI_WINDOW_DAYS,
    responseRate: { handled, received, ratePct: ratePct(handled, received) },
    averageStarRating: round(starRows[0]?.avg ?? null, 2),
    reviewVolume: { currentMonth: current, priorMonth: prior, deltaPct: deltaPct(current, prior) },
    averageTimeToRespondHours: round(ttrRows[0]?.avg_hours ?? null, 1),
    sentimentTrend: mapTrend(trendRows),
  };
}
```

```ts
    scope,
    windowDays: KPI_WINDOW_DAYS,
    responseRate: { handled, received, ratePct: ratePct(handled, received) },
    averageStarRating: round(starRows[0]?.avg ?? null, 2),
    reviewVolume: { currentMonth: current, priorMonth: prior, deltaPct: deltaPct(current, prior) },
    averageTimeToRespondHours: round(ttrRows[0]?.avg_hours ?? null, 1),
    sentimentTrend: mapTrend(trendRows),
  };
}
```
