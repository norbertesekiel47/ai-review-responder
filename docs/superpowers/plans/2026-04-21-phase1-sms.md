# Phase 1 SMS — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add SMS as a verified peer of email in RepWorth's approval-notification dispatcher — enrollment via Twilio Verify, notification delivery via Twilio Messaging Service, and TCPA-compliant STOP/START handling — all behind a `CHANNELS_SMS_ENABLED` env flag.

**Architecture:** `SmsChannel` implements the existing Phase 0 `NotificationChannel` interface. Sender-level opt-out state lives in a new `sms_sender_opt_outs` table (keyed by `(messaging_service_sid, handle)`) with an append-only `sms_sender_opt_out_events` audit log. `NotificationWorker` dispatcher adds cost reservation (Redis) + refund via try/finally around the send. Admin widget (server-gated by flag + subscription) handles enrollment lifecycle through 6 routes with advisory-lock serialization.

**Tech Stack:** Next.js 14 App Router (admin routes), Fastify (worker ingress), Postgres.js with `sql.begin` transactions, Upstash Redis (cost guard), Twilio SDK (Verify + Messaging Service), libphonenumber-js, @sindresorhus/transliterate, Vitest (unit + integration).

**Spec:** `docs/superpowers/specs/2026-04-21-phase1-sms-design.md`

**Prerequisite branch:** `feat/communication-channels` (Phase 0 shipped). All work lands on this branch.

```sql
-- supabase/migrations/022_sms_channel_infra.sql
--
-- Phase 1 SMS infrastructure. Bundles Phase 0's deferred column drop with
-- Phase 1's soft-delete support, sender-level opt-out tracking, audit log,
-- and the inbound-webhook handle index.
--
-- Ordering: ADD all columns BEFORE creating indexes that reference them.

-- 1. Drop deprecated column (promised in migration 021). No data loss —
-- commit 1b5d40e removed the last read/write.
ALTER TABLE businesses DROP COLUMN IF EXISTS notification_preference;

-- 2. Soft-delete columns on channel_enrollments. /delete sets archived_at
-- rather than hard-deleting, preserving TCPA consent/opt-out evidence.
ALTER TABLE channel_enrollments
  ADD COLUMN IF NOT EXISTS archived_at TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS archive_reason TEXT;

-- 3a. sms_sender_opt_outs: authoritative CURRENT sender-level opt-out
-- state. Keyed by (messaging_service_sid, handle) so future multi-MS
-- setups (staging/prod split, sender migration) scope opt-outs correctly.
-- Persists across enrollment lifecycles — Twilio's actual block state
-- is never out of sync with our DB.
CREATE TABLE IF NOT EXISTS sms_sender_opt_outs (
  messaging_service_sid  TEXT NOT NULL,
  handle                 TEXT NOT NULL,
  opted_out_at           TIMESTAMPTZ NOT NULL,
  opted_in_at            TIMESTAMPTZ,
  last_event_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  PRIMARY KEY (messaging_service_sid, handle)
);

-- 3b. sms_sender_opt_out_events: append-only audit log of every STOP,
-- START, and send-time 21610 event. Durable TCPA evidence independent
-- of inbound_events 30-day archival.
CREATE TABLE IF NOT EXISTS sms_sender_opt_out_events (
  id                      UUID PRIMARY KEY,   -- uuidv7 (app-generated)
  messaging_service_sid   TEXT NOT NULL,
  handle                  TEXT NOT NULL,
  event_type              TEXT NOT NULL,   -- 'stop' | 'start' | '21610_at_send'
  source                  TEXT NOT NULL,   -- 'twilio_inbound_keyword' | 'twilio_optouttype' | 'twilio_21610'
  inbound_event_id        UUID REFERENCES inbound_events(id) ON DELETE SET NULL,
  created_at              TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_sms_sender_opt_out_events_handle
  ON sms_sender_opt_out_events(messaging_service_sid, handle, created_at DESC);

-- 3c. RLS on both new tables. No policies defined → service_role only.
ALTER TABLE sms_sender_opt_outs ENABLE ROW LEVEL SECURITY;
ALTER TABLE sms_sender_opt_out_events ENABLE ROW LEVEL SECURITY;

-- 4. Replace UNIQUE(business_id, channel) with a partial unique INDEX
-- scoped to active (non-archived) rows.
--
-- COORDINATED CODE CHANGE REQUIRED: every ON CONFLICT (business_id, channel)
-- DO UPDATE site must include the WHERE archived_at IS NULL predicate
-- (see Task 3).
ALTER TABLE channel_enrollments
  DROP CONSTRAINT IF EXISTS channel_enrollments_business_id_channel_key;
CREATE UNIQUE INDEX IF NOT EXISTS channel_enrollments_business_channel_active_key
  ON channel_enrollments (business_id, channel)
  WHERE archived_at IS NULL;

-- 5. Inbound webhook handle lookup index, scoped to ACTIVE rows only.
-- STOP/START handlers never act on archived rows.
CREATE INDEX IF NOT EXISTS idx_channel_enrollments_sms_handle
  ON channel_enrollments(handle)
  WHERE channel IN ('sms', 'whatsapp') AND handle IS NOT NULL AND archived_at IS NULL;
```

```typescript
// src/lib/notifications/sms-body.ts
import transliterate from '@sindresorhus/transliterate';

export class SmsBodyTooLongError extends Error {
  constructor(public readonly length: number) {
    super(`sms_body_too_long:${length}`);
    this.name = 'SmsBodyTooLongError';
  }
}

export interface FormatApprovalSmsParams {
  reviewerName: string;
  starRating: number;
  approvalUrl: string;
}

const GSM7_REGEX = /^[\x00-\x7F\n]+$/;

function formatReviewerName(raw: string): string {
  const trimmed = raw.trim().replace(/\s+/g, ' ');
  if (trimmed.length === 0) return 'a reviewer';
  const ascii = transliterate(trimmed).replace(/[^\x00-\x7F]/g, '').trim();
  if (ascii.length === 0) return 'a reviewer';
  const parts = ascii.split(' ');
  const first = parts[0] ?? '';
  const lastInitial = parts.length > 1 ? `${(parts[parts.length - 1] ?? '').charAt(0)}.` : '';
  const formatted = lastInitial ? `${first} ${lastInitial}` : first;
  if (formatted.length === 0) return 'a reviewer';
  if (formatted.length > 29) return formatted.slice(0, 25) + '..';
  return formatted;
}

export function formatApprovalSms(p: FormatApprovalSmsParams): string {
  const formattedName = formatReviewerName(p.reviewerName);
  const tier1 = `New ${p.starRating}-star review from ${formattedName}. Draft ready: ${p.approvalUrl}\nReply STOP to unsubscribe.`;
  if (tier1.length <= 160) {
    if (!GSM7_REGEX.test(tier1)) throw new SmsBodyTooLongError(tier1.length); // defensive
    return tier1;
  }
  const tier2 = `New ${p.starRating}-star review. Draft ready: ${p.approvalUrl}\nReply STOP to unsubscribe.`;
  if (tier2.length <= 160) {
    if (!GSM7_REGEX.test(tier2)) throw new SmsBodyTooLongError(tier2.length);
    return tier2;
  }
  throw new SmsBodyTooLongError(tier2.length);
}
```

```typescript
function todayDateStr(): string { return new Date().toISOString().slice(0, 10); }
function perBizKey(bId: string, d: string): string { return `cost:sms:${bId}:${d}`; }
function globalKey(d: string): string { return `cost:sms:global:${d}`; }

function parseCentsEnv(name: string, fallback: number): number {
  const val = process.env[name];
  if (!val) return fallback;
  const trimmed = val.trim();
  if (!/^\d+$/.test(trimmed)) {
    console.warn(`[sms-tracker] Invalid env ${name}=${val} — using fallback ${fallback}`);
    return fallback;
  }
  const n = Number(trimmed);
  if (!Number.isSafeInteger(n) || n < 0) return fallback;
  return n;
}

export async function reserveSmsCost(businessId: string): Promise<string> {
  const redis = getCacheRedis();
  const date = todayDateStr();
  const bKey = perBizKey(businessId, date);
  const gKey = globalKey(date);

  const results = await redis.pipeline()
    .incrby(bKey, SMS_SEND_COST_CENTS)
    .incrby(gKey, SMS_SEND_COST_CENTS)
    .exec();
  const bizTotal = results![0]![1] as number;
  const globalTotal = results![1]![1] as number;

  const perBizCap = parseCentsEnv('SMS_DAILY_LIMIT_CENTS_PER_BUSINESS', 10);
  const globalCap = parseCentsEnv('SMS_DAILY_LIMIT_CENTS_GLOBAL', 500);

  if (bizTotal > perBizCap || globalTotal > globalCap) {
    await redis.pipeline().decrby(bKey, SMS_SEND_COST_CENTS).decrby(gKey, SMS_SEND_COST_CENTS).exec();
    throw new DailySmsLimitExceededError(bizTotal > perBizCap ? 'per_business' : 'global');
  }
  await redis.pipeline().expire(bKey, TTL_SECONDS).expire(gKey, TTL_SECONDS).exec();
  return date;
}

export async function refundSmsCost(businessId: string, reservationDate: string): Promise<void> {
  const redis = getCacheRedis();
  await redis.pipeline()
    .decrby(perBizKey(businessId, reservationDate), SMS_SEND_COST_CENTS)
    .decrby(globalKey(reservationDate), SMS_SEND_COST_CENTS)
    .exec();
}
```

```typescript
// worker/ingress/twilio.ts
import type { FastifyRequest, FastifyReply } from 'fastify';
import { v7 as uuidv7 } from 'uuid';
import { sql, jsonb } from '../../src/lib/db/server';
import { parseTwilioBody, verifyTwilioSignature } from './twilio-verify';

const LOG_PREFIX = '[ingress/twilio]';

function fullUrl(request: FastifyRequest): string {
  return `${process.env.WORKER_PUBLIC_URL}${request.url}`;
}

async function claimInboundEvent(dedupKey: string, eventType: string, payload: unknown): Promise<'claimed' | 'duplicate'> {
  const inserted = await sql`
    INSERT INTO inbound_events (id, source, external_event_key, event_type, payload, status)
    VALUES (${uuidv7()}, 'twilio', ${dedupKey}, ${eventType}, ${jsonb(payload)}, 'received')
    ON CONFLICT (external_event_key) DO NOTHING
  `;
  if (inserted.count > 0) return 'claimed';
  const reclaimed = await sql<{ id: string }[]>`
    UPDATE inbound_events
    SET received_at = NOW(), status = 'received', error_message = NULL
    WHERE external_event_key = ${dedupKey}
      AND (status = 'failed' OR (status = 'received' AND received_at < NOW() - INTERVAL '2 minutes'))
    RETURNING id
  `;
  return reclaimed.length > 0 ? 'claimed' : 'duplicate';
}

export async function handleTwilioStatusCallback(request: FastifyRequest, reply: FastifyReply): Promise<void> {
  const rawBody = (request as unknown as { rawBody?: string }).rawBody ?? '';
  const params = parseTwilioBody(rawBody);
  const sig = request.headers['x-twilio-signature'] as string | undefined;

  if (!verifyTwilioSignature({ authToken: process.env.TWILIO_AUTH_TOKEN, signatureHeader: sig, fullUrl: fullUrl(request), params })) {
    reply.status(401).send({ error: 'invalid_signature' });
    return;
  }

  const { MessageSid, MessageStatus, ErrorCode } = params;
  if (!MessageSid || !MessageStatus) { reply.status(400).send({ error: 'missing_fields' }); return; }

  const dedupKey = `${MessageSid}:${MessageStatus}`;
  const claim = await claimInboundEvent(dedupKey, `status:${MessageStatus}`, params);
  if (claim === 'duplicate') { reply.status(200).send({ received: true, duplicate: true }); return; }

  const isTerminal = ['delivered', 'undelivered', 'failed'].includes(MessageStatus);

  if (!isTerminal) {
    if (!['queued', 'sending', 'sent'].includes(MessageStatus)) {
      console.log(`${LOG_PREFIX} Unknown MessageStatus ${MessageStatus} for ${MessageSid}`);
    }
    await sql`UPDATE inbound_events SET status='processed', processed_at=NOW() WHERE external_event_key = ${dedupKey}`;
    reply.status(200).send({ received: true });
    return;
  }

  const [row] = await sql<{ id: string; delivered_at: Date | null; failed_at: Date | null }[]>`
    SELECT id, delivered_at, failed_at FROM notification_deliveries
    WHERE external_message_id = ${MessageSid} AND channel = 'sms' LIMIT 1
  `;
  if (!row) {
    await sql`UPDATE inbound_events SET status='failed', error_message='delivery_row_not_yet_present' WHERE external_event_key = ${dedupKey}`;
    reply.status(503).send({ received: true, retry: 'delivery_row_not_yet_present' });
    return;
  }
  if (row.delivered_at || row.failed_at) {
    await sql`UPDATE inbound_events SET status='processed', processed_at=NOW() WHERE external_event_key = ${dedupKey}`;
    reply.status(200).send({ received: true, terminal_state_preserved: true });
    return;
  }

  if (MessageStatus === 'delivered') {
    await sql`UPDATE notification_deliveries SET delivered_at = NOW() WHERE id = ${row.id} AND channel = 'sms' AND delivered_at IS NULL AND failed_at IS NULL`;
  } else {
    await sql`UPDATE notification_deliveries SET failed_at = NOW(), failure_reason = ${`twilio:${ErrorCode ?? 'unknown'}`} WHERE id = ${row.id} AND channel = 'sms' AND delivered_at IS NULL AND failed_at IS NULL`;
  }
  await sql`UPDATE inbound_events SET status='processed', processed_at=NOW() WHERE external_event_key = ${dedupKey}`;
  reply.status(200).send({ received: true });
}
```

```typescript
// append to worker/ingress/twilio.ts

const STOP_KEYWORDS = new Set([
  'STOP', 'STOPALL', 'UNSUBSCRIBE', 'CANCEL', 'END', 'QUIT', 'REVOKE', 'OPTOUT',
  'PARE', 'PARAR', 'PARARLO', 'CANCELAR', 'DESUSCRIBIR', 'DETENER',
]);
const START_KEYWORDS = new Set(['START', 'UNSTOP', 'YES', 'ARRANCAR']);

function classifyIntent(params: Record<string, string>): { intent: 'stop' | 'start' | 'none'; source: string } {
  const optOutType = params.OptOutType;
  if (optOutType === 'STOP') return { intent: 'stop', source: 'twilio_optouttype' };
  if (optOutType === 'START' || optOutType === 'UNSTOP') return { intent: 'start', source: 'twilio_optouttype' };
  const body = (params.Body ?? '').trim().toUpperCase().replace(/^[^A-Z]+|[^A-Z]+$/g, '');
  const first = body.split(/\s+/)[0] ?? '';
  if (STOP_KEYWORDS.has(first)) return { intent: 'stop', source: 'twilio_inbound_keyword' };
  if (START_KEYWORDS.has(first)) return { intent: 'start', source: 'twilio_inbound_keyword' };
  return { intent: 'none', source: '' };
}

function redactedPayload(params: Record<string, string>): Record<string, string> {
  return { ...params, Body: '[redacted]' };
}

export async function handleTwilioInbound(request: FastifyRequest, reply: FastifyReply): Promise<void> {
  const rawBody = (request as unknown as { rawBody?: string }).rawBody ?? '';
  const params = parseTwilioBody(rawBody);
  const sig = request.headers['x-twilio-signature'] as string | undefined;

  if (!verifyTwilioSignature({ authToken: process.env.TWILIO_AUTH_TOKEN, signatureHeader: sig, fullUrl: fullUrl(request), params })) {
    reply.status(401).send({ error: 'invalid_signature' });
    return;
  }

  const { MessageSid, From, MessagingServiceSid } = params;
  if (!MessageSid || !From) { reply.status(400).send({ error: 'missing_fields' }); return; }
  const msid = MessagingServiceSid || process.env.TWILIO_MESSAGING_SERVICE_SID!;

  const dedupKey = MessageSid;
  const claim = await claimInboundEvent(dedupKey, 'inbound_sms', redactedPayload(params));
  if (claim === 'duplicate') { reply.status(200).send({ received: true, duplicate: true }); return; }

  const [inboundRow] = await sql<{ id: string }[]>`SELECT id FROM inbound_events WHERE external_event_key = ${dedupKey} LIMIT 1`;
  const inboundEventId = inboundRow?.id ?? null;

  const { intent, source } = classifyIntent(params);

  try {
    if (intent === 'stop') {
      await sql.begin(async (tx) => {
        await tx`
          INSERT INTO sms_sender_opt_outs (messaging_service_sid, handle, opted_out_at, opted_in_at, last_event_at)
          VALUES (${msid}, ${From}, NOW(), NULL, NOW())
          ON CONFLICT (messaging_service_sid, handle) DO UPDATE
            SET opted_out_at = EXCLUDED.opted_out_at,
                opted_in_at = NULL,
                last_event_at = EXCLUDED.last_event_at
        `;
        await tx`
          INSERT INTO sms_sender_opt_out_events (id, messaging_service_sid, handle, event_type, source, inbound_event_id)
          VALUES (${uuidv7()}, ${msid}, ${From}, 'stop', ${source}, ${inboundEventId})
        `;
        await tx`
          UPDATE channel_enrollments
          SET status = 'unsubscribed',
              unsubscribed_at = COALESCE(unsubscribed_at, NOW()),
              unsubscribed_reason = COALESCE(unsubscribed_reason, 'sms_stop'),
              updated_at = NOW()
          WHERE handle = ${From} AND channel = 'sms' AND archived_at IS NULL
        `;
      });
    } else if (intent === 'start') {
      await sql.begin(async (tx) => {
        await tx`
          UPDATE sms_sender_opt_outs
          SET opted_in_at = NOW(), last_event_at = NOW()
          WHERE messaging_service_sid = ${msid} AND handle = ${From} AND opted_out_at IS NOT NULL
        `;
        await tx`
          INSERT INTO sms_sender_opt_out_events (id, messaging_service_sid, handle, event_type, source, inbound_event_id)
          VALUES (${uuidv7()}, ${msid}, ${From}, 'start', ${source}, ${inboundEventId})
        `;
      });
    }
    await sql`UPDATE inbound_events SET status='processed', processed_at=NOW() WHERE external_event_key = ${dedupKey}`;
    reply.status(200).send({ received: true });
  } catch (err) {
    const msg = err instanceof Error ? err.message : String(err);
    console.error(`${LOG_PREFIX} inbound failed: ${msg}`);
    await sql`UPDATE inbound_events SET status='failed', error_message=${msg} WHERE external_event_key = ${dedupKey}`;
    reply.status(503).send({ received: true, retry: true });
  }
}
```

```typescript
  if (status !== 'active') return subscriptionNotActive();

  return withSmsEnrollmentLock(business_id, async (tx) => {
    // Step 1: Load current active row
    const [row] = await tx<{ handle: string; status: string }[]>`
      SELECT handle, status FROM channel_enrollments
      WHERE business_id = ${business_id} AND channel = 'sms' AND archived_at IS NULL
    `;

    // Step 2: Resolve twilioPhone + input validation
    let twilioPhone: string;
    let dbAction: 'insert' | 'update' | 'noop';
    if (!row) {
      if (!phone) return BAD_REQUEST('phone_required');
      try {
        const parsed = parsePhoneNumber(phone, 'US');
        if (!parsed || !parsed.isValid() || parsed.countryCallingCode !== '1') return BAD_REQUEST('phone_invalid');
        twilioPhone = parsed.number;
      } catch { return BAD_REQUEST('phone_invalid'); }
      dbAction = 'insert';
    } else {
      if (phone && phone !== row.handle) return BAD_REQUEST('handle_change_forbidden');
      twilioPhone = row.handle;
      if (row.status === 'verified') return BAD_REQUEST('already_verified');
      if (row.status === 'pending_verification') { dbAction = 'noop'; }
      else { dbAction = 'update'; }  // disabled | unsubscribed
    }

    // Step 3: Sender-opt-out gate (scoped to Messaging Service SID)
    const [optOut] = await tx<{ opted_out_at: Date; opted_in_at: Date | null }[]>`
      SELECT opted_out_at, opted_in_at FROM sms_sender_opt_outs
      WHERE messaging_service_sid = ${process.env.TWILIO_MESSAGING_SERVICE_SID!}
        AND handle = ${twilioPhone}
    `;
    if (optOut && optOut.opted_out_at && (!optOut.opted_in_at || optOut.opted_in_at < optOut.opted_out_at)) {
      return BAD_REQUEST('sender_opt_in_required');
    }

    // Step 4: Twilio Verify (before DB mutation)
    const client = twilio(process.env.TWILIO_ACCOUNT_SID!, process.env.TWILIO_AUTH_TOKEN!);
    try {
      await client.verify.v2.services(process.env.TWILIO_VERIFY_SERVICE_SID!).verifications.create({ to: twilioPhone, channel: 'sms' });
    } catch (err) {
      console.error(`[sms/verify] Twilio Verify failed: ${err instanceof Error ? err.message : err}`);
      return TWILIO_VERIFY_ERROR('verify_send_failed');
    }

    // Step 5: DB mutation
    if (dbAction === 'insert') {
      await tx`
        INSERT INTO channel_enrollments (id, business_id, channel, handle, status)
        VALUES (${uuidv7()}, ${business_id}, 'sms', ${twilioPhone}, 'pending_verification')
      `;
    } else if (dbAction === 'update') {
      const result = await tx<{ id: string }[]>`
        UPDATE channel_enrollments
        SET status = 'pending_verification', updated_at = NOW()
        WHERE business_id = ${business_id} AND channel = 'sms' AND archived_at IS NULL
        RETURNING id
      `;
      if (result.length !== 1) return NextResponse.json({ ok: false, reason: 'race_conflict' }, { status: 409 });
    }

    const masked = maskPhone(twilioPhone);
    return NextResponse.json({ ok: true, phoneMasked: masked });
  });
}

function maskPhone(e164: string): string {
  const rest = e164.replace(/^\+1/, '');
  if (rest.length !== 10) return '+1 XXX-XXX-XXXX';
  return `+1 ${rest.slice(0, 3)}-XXX-${rest.slice(6)}`;
}
```

```typescript
  try { requireSmsEnabled(); } catch { return flagOff(); }
  const session = await requireAdminSession(request);
  if (!session) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });

      SELECT handle, status FROM channel_enrollments
      WHERE business_id = ${business_id} AND channel = 'sms' AND archived_at IS NULL
    `;
    if (!row) return BAD_REQUEST('no_pending_verification');

    // Step 2: Re-check sender-opt-out gate
    const [optOut] = await tx<{ opted_out_at: Date; opted_in_at: Date | null }[]>`
      SELECT opted_out_at, opted_in_at FROM sms_sender_opt_outs
      WHERE messaging_service_sid = ${process.env.TWILIO_MESSAGING_SERVICE_SID!}
        AND handle = ${row.handle}
    `;
    if (optOut && optOut.opted_out_at && (!optOut.opted_in_at || optOut.opted_in_at < optOut.opted_out_at)) {
      return CONFLICT('sender_opted_out_during_verification');
    }

    // Step 3: Dispatch on status
    if (row.status !== 'pending_verification') return CONFLICT('row_not_pending');

    // Step 4: Twilio Verify check
    const client = twilio(process.env.TWILIO_ACCOUNT_SID!, process.env.TWILIO_AUTH_TOKEN!);
    const check = await client.verify.v2.services(process.env.TWILIO_VERIFY_SERVICE_SID!).verificationChecks.create({ to: row.handle, code });

    // Step 5: On approved, UPDATE to verified
    if (check.status !== 'approved') return BAD_REQUEST('invalid_code_or_expired');

    const result = await tx<{ id: string }[]>`
      UPDATE channel_enrollments
      SET status = 'verified',
          verified_at = NOW(),
          unsubscribed_at = NULL,
          unsubscribed_reason = NULL,
          updated_at = NOW()
      WHERE business_id = ${business_id} AND channel = 'sms'
        AND status = 'pending_verification' AND archived_at IS NULL
      RETURNING id
    `;
    if (result.length !== 1) return CONFLICT('race_conflict');

    return NextResponse.json({ ok: true });
  });
}
```

```typescript
// src/app/api/admin/channels/sms/status/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { requireAdminSession } from '@/lib/auth/admin';
import { flagOff, requireSmsEnabled } from '../_shared';
import { sql } from '@/lib/db/server';

export async function GET(request: NextRequest): Promise<NextResponse> {
  try { requireSmsEnabled(); } catch { return flagOff(); }
  const session = await requireAdminSession(request);
  if (!session) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });

  const businessId = new URL(request.url).searchParams.get('business_id');
  if (!businessId) return NextResponse.json({ error: 'business_id_required' }, { status: 400 });

  const [row] = await sql<Array<{
    status: string; handle: string;
    verified_at: Date | null; disabled_at: Date | null;
    unsubscribed_at: Date | null; unsubscribed_reason: string | null;
    archived_at: Date | null;
    sender_opt_in_cleared: boolean;
  }>>`
    SELECT ce.status, ce.handle, ce.verified_at, ce.disabled_at,
           ce.unsubscribed_at, ce.unsubscribed_reason, ce.archived_at,
           CASE
             WHEN so.handle IS NULL THEN TRUE
             WHEN so.opted_in_at IS NOT NULL AND so.opted_in_at > so.opted_out_at THEN TRUE
             ELSE FALSE
           END AS sender_opt_in_cleared
    FROM channel_enrollments ce
    LEFT JOIN sms_sender_opt_outs so
      ON so.handle = ce.handle
     AND so.messaging_service_sid = ${process.env.TWILIO_MESSAGING_SERVICE_SID!}
    WHERE ce.business_id = ${businessId} AND ce.channel = 'sms' AND ce.archived_at IS NULL
    LIMIT 1
  `;

  return NextResponse.json({ enrollment: row ?? null });
}
```

```tsx
// src/components/admin/SmsEnrollmentWidget.tsx
'use client';

import { useEffect, useState } from 'react';

interface EnrollmentState {
  status: 'pending_verification' | 'verified' | 'disabled' | 'unsubscribed';
  handle: string;
  sender_opt_in_cleared: boolean;
}

type WidgetState =
  | { kind: 'empty' }
  | { kind: 'sent'; maskedPhone: string }
  | { kind: 'verified'; maskedPhone: string }
  | { kind: 'disabled'; maskedPhone: string }
  | { kind: 'unsubscribed'; maskedPhone: string; reason: string | null }
  | { kind: 'awaiting_sender_opt_in'; maskedPhone: string }
  | { kind: 'opt_in_received'; maskedPhone: string };

function maskPhoneForDisplay(e164: string): string {
  const rest = e164.replace(/^\+1/, '');
  if (rest.length !== 10) return e164;
  return `+1 ${rest.slice(0, 3)}-XXX-${rest.slice(6)}`;
}

function deriveState(enrollment: EnrollmentState | null, uiHint: 'waiting' | null): WidgetState {
  if (!enrollment) return { kind: 'empty' };
  const masked = maskPhoneForDisplay(enrollment.handle);
  if (enrollment.status === 'pending_verification') return { kind: 'sent', maskedPhone: masked };
  if (enrollment.status === 'verified') return { kind: 'verified', maskedPhone: masked };
  if (enrollment.status === 'disabled') return { kind: 'disabled', maskedPhone: masked };
  // unsubscribed — depends on sender_opt_in_cleared + UI waiting flag
  if (uiHint === 'waiting') {
    return enrollment.sender_opt_in_cleared
      ? { kind: 'opt_in_received', maskedPhone: masked }
      : { kind: 'awaiting_sender_opt_in', maskedPhone: masked };
  }
  return { kind: 'unsubscribed', maskedPhone: masked, reason: null };
}

export function SmsEnrollmentWidget({ businessId }: { businessId: string }) {
  const [enrollment, setEnrollment] = useState<EnrollmentState | null>(null);
  const [uiHint, setUiHint] = useState<'waiting' | null>(null);
  const [phoneInput, setPhoneInput] = useState('+1');
  const [codeInput, setCodeInput] = useState('');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function loadStatus() {
    const res = await fetch(`/api/admin/channels/sms/status?business_id=${businessId}`, { method: 'GET' });
    const json = await res.json();
    setEnrollment(json.enrollment);
  }
  useEffect(() => { void loadStatus(); }, [businessId]);

  // Polling for awaiting_sender_opt_in state (every 3s)
  useEffect(() => {
    if (uiHint !== 'waiting') return;
    const interval = setInterval(loadStatus, 3000);
    return () => clearInterval(interval);
  }, [uiHint, businessId]);

  const state = deriveState(enrollment, uiHint);

  async function call(path: string, body: unknown): Promise<any> {
    setBusy(true); setError(null);
    try {
      const res = await fetch(`/api/admin/channels/sms/${path}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
      });
      const json = await res.json();
      if (!res.ok) { setError(json.reason ?? 'error'); return null; }
      return json;
    } finally { setBusy(false); }
  }

  async function onSendCode() {
    const body = state.kind === 'empty' ? { business_id: businessId, phone: phoneInput } : { business_id: businessId };
    const ok = await call('verify', body);
    if (ok) await loadStatus();
  }
  async function onConfirm() {
    const ok = await call('confirm', { business_id: businessId, code: codeInput });
    if (ok) { setCodeInput(''); await loadStatus(); }
  }
  async function onCancel() {
    await call('cancel', { business_id: businessId });
    await loadStatus();
  }
  async function onDisable() {
    await call('disable', { business_id: businessId });
    await loadStatus();
  }
  async function onDelete() {
    if (!confirm('This soft-deletes the enrollment while preserving audit history. A new enrollment can then be created with a different phone number. Continue?')) return;
    await call('delete', { business_id: businessId, confirm: true });
    setUiHint(null);
    await loadStatus();
  }

  return (
    <div className="sms-enrollment-widget">
      {/* Render branches for each state.kind — full JSX omitted for brevity in plan,
          but each state maps 1:1 to the state machine in spec §8.1.
          Key buttons:
            empty → [E.164 input] [Send code]
            sent → [6-digit code input] [Verify] [Cancel]
            verified → ✅ {masked}, [Disable] [Delete]
            disabled → ⛔ {masked}, [Re-enable (send to {masked})] [Delete]
            unsubscribed → 🚫 {masked}, [Start re-enable] [Delete]
            awaiting_sender_opt_in → instructions + polling spinner + [Cancel waiting]
            opt_in_received → "✅ Opt-in detected." [Send verification code] [Cancel]
      */}
      {state.kind === 'empty' && (
        <div>
          <input value={phoneInput} onChange={(e) => setPhoneInput(e.target.value)} placeholder="+1 555 555 5555" />
          <button onClick={onSendCode} disabled={busy}>Send code</button>
        </div>
      )}
      {state.kind === 'sent' && (
        <div>
          <p>Code sent to {state.maskedPhone}</p>
          <input value={codeInput} onChange={(e) => setCodeInput(e.target.value)} placeholder="123456" />
          <button onClick={onConfirm} disabled={busy}>Verify</button>
          <button onClick={onCancel} disabled={busy}>Cancel</button>
        </div>
      )}
      {state.kind === 'verified' && (
        <div>
          <p>✅ SMS: {state.maskedPhone}</p>
          <button onClick={onDisable} disabled={busy}>Disable</button>
          <button onClick={onDelete} disabled={busy}>Delete enrollment</button>
        </div>
      )}
      {state.kind === 'disabled' && (
        <div>
          <p>⛔ SMS disabled: {state.maskedPhone}</p>
          <button onClick={onSendCode} disabled={busy}>Re-enable — send code to {state.maskedPhone}</button>
          <button onClick={onDelete} disabled={busy}>Delete enrollment</button>
        </div>
      )}
      {state.kind === 'unsubscribed' && (
        <div>
          <p>🚫 SMS unsubscribed: {state.maskedPhone}</p>
          <button onClick={() => setUiHint('waiting')} disabled={busy}>Start re-enable</button>
          <button onClick={onDelete} disabled={busy}>Delete enrollment</button>
        </div>
      )}
      {state.kind === 'awaiting_sender_opt_in' && (
        <div>
          <p>Ask the owner at {state.maskedPhone} to text <b>START</b> to our RepWorth number. We'll detect it automatically.</p>
          <button onClick={() => setUiHint(null)} disabled={busy}>Cancel waiting</button>
        </div>
      )}
      {state.kind === 'opt_in_received' && (
        <div>
          <p>✅ Opt-in detected for {state.maskedPhone}. Send the verification code now.</p>
          <button onClick={async () => { await onSendCode(); setUiHint(null); }} disabled={busy}>Send verification code</button>
          <button onClick={() => setUiHint(null)} disabled={busy}>Cancel</button>
        </div>
      )}
      {error && <p className="text-red-600">Error: {error}</p>}
    </div>
  );
}
```
